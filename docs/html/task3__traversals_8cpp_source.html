12static void inRec(const Node *n, std::string &out) {
13 if (!n) return;

15 out += n->val;
16 } else if (n->val == "~") { // unary: (~operand)

35std::string inOrderIterative(const Node *root) {
36 // Work item: either "expand this node" (n != nullptr) or "emit this literal" (n == nullptr).
37 struct Item { const Node *n; const char *lit; };
38 std::string out;
39 if (!root) return out;
40 std::vector<Item> st;
41 st.push_back({root, nullptr});
42 while (!st.empty()) {
43 Item it = st.back();
44 st.pop_back();
45 if (!it.n) {
46 if (it.lit) out += it.lit;
47 continue;
48 }
49 const Node *n = it.n;
50 if (!n->left && !n->right) {
51 out += n->val;
52 } else if (n->val == "~") {
53 // want: ( ~ left ) -> push in reverse order
54 st.push_back({nullptr, ")"});
55 st.push_back({n->left, nullptr});
56 st.push_back({nullptr, "~"});
57 st.push_back({nullptr, "("});
58 } else {
59 // want: ( left op right )
60 st.push_back({nullptr, ")"});
61 st.push_back({n->right, nullptr});
62 st.push_back({nullptr, n->val.c_str()});
63 st.push_back({n->left, nullptr});
64 st.push_back({nullptr, "("});
65 }
66 }
67 return out;

115std::vector<const Node *> postOrderNodes(const Node *root) {
116 std::vector<const Node *> order, st;
117 const Node *cur = root, *last = nullptr;
118 while (cur || !st.empty()) {
119 if (cur) {
120 st.push_back(cur);
121 cur = cur->left;
122 } else {
123 const Node *top = st.back();
124 if (top->right && last != top->right) {
125 cur = top->right; // go into the right subtree first
126 } else {
127 order.push_back(top); // both subtrees done -> visit
128 last = top;
129 st.pop_back();
130 }
131 }
132 }
133 return order;

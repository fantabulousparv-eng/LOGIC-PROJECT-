76 std::vector<std::string> positional;

84 else if (a == "--no-recursive") useRecursive = false;

88 std::vector<FileMetrics> results;

90 if (mode == "demo") {
91 runConsoleDemo("((p+q)*(~r))");
92 runConsoleDemo("((p>q)+(~(q*r)))");
93 return 0;
94 } else if (mode == "formula") {
95 if (positional.empty()) { usage(); return 1; }
96 runConsoleDemo(positional[0]);
97 if (!reportPath.empty()) results.push_back(analyseFormula("formula", positional[0], maxVars, useRecursive));
98 } else if (mode == "cnf" || mode == "dir") {
99 std::vector<std::string> files;
100 if (mode == "cnf") {
101 files = positional;
102 } else {
103 if (positional.empty()) { usage(); return 1; }
104 for (const auto &e : fs::recursive_directory_iterator(positional[0]))
105 if (e.is_regular_file() && e.path().extension() == ".cnf") files.push_back(e.path().string());
106 std::sort(files.begin(), files.end());
107 }
108 if (files.empty()) { std::cerr << "no .cnf files given/found\n"; return 1; }
109 if (reportPath.empty()) reportPath = "report.html";
110 for (const std::string &f : files) {
111 std::cout << "Analysing " << f << " ... " << std::flush;
112 try {
113 CnfInfo cnf = readDimacs(f);
114 std::string infix = cnfToInfix(cnf);
115 FileMetrics m = analyseFormula(fs::path(f).filename().string(), infix, maxVars, useRecursive);
116 m.numClauses = cnf.numClauses;
117 std::cout << (m.ok ? "ok" : "FAILED: " + m.error) << "\n";
118 results.push_back(m);
119 } catch (const std::exception &e) {
120 FileMetrics m;
121 m.name = fs::path(f).filename().string();
122 m.error = e.what();
123 std::cout << "FAILED: " << e.what() << "\n";
124 results.push_back(m);
125 }
126 }
127 } else {
128 usage();
129 return 1;
130 }
131 } catch (const std::exception &e) {
132 std::cerr << "Error: " << e.what() << "\n";
133 return 1;
134 }
135
136 if (!reportPath.empty()) {
137 writeHtmlReport(results, reportPath);
138 std::cout << "HTML report written to " << reportPath << "\n";
139 }
140 return 0;

8std::vector<std::string> collectAtoms(Node *root) {
9 std::vector<std::string> names;
10 std::unordered_map<std::string, int> index;
11 if (!root) return names;
12 std::vector<Node *> st;
13 st.push_back(root);
14 while (!st.empty()) { // pre-order => atoms numbered left to right
15 Node *n = st.back();
16 st.pop_back();
17 if (!opSymbol(n)) {
18 auto it = index.find(n->val);
19 if (it == index.end()) {
20 it = index.emplace(n->val, static_cast<int>(names.size())).first;
21 names.push_back(n->val);
22 }
23 n->atomIdx = it->second;
24 }
25 if (n->right) st.push_back(n->right);
26 if (n->left) st.push_back(n->left);
27 }
28 return names;
29}

40bool evaluateRecursive(const Node *n, const std::vector<char> &vals) {
41 char op = opSymbol(n);
42 if (!op) {
43 if (n->atomIdx < 0) throw std::logic_error("atoms not indexed - call collectAtoms() first");
44 return vals[n->atomIdx] != 0;
45 }
46 if (op == '~') return !evaluateRecursive(n->left, vals);
47 bool a = evaluateRecursive(n->left, vals);
48 bool b = evaluateRecursive(n->right, vals);
49 return apply(op, a, b);
50}

52bool evaluateOrdered(const std::vector<const Node *> &order, const std::vector<char> &vals, std::vector<char> &stack) {
53 stack.clear();
54 for (const Node *n : order) {
55 char op = opSymbol(n);
56 if (!op) {
57 if (n->atomIdx < 0) throw std::logic_error("atoms not indexed - call collectAtoms() first");
58 stack.push_back(vals[n->atomIdx]);
59 } else if (op == '~') {
60 stack.back() = !stack.back();
61 } else {
62 bool b = stack.back() != 0; stack.pop_back();
63 bool a = stack.back() != 0; stack.pop_back();
64 stack.push_back(apply(op, a, b));
65 }
66 }
67 return stack.back() != 0;
68}

75TruthTableResult truthTable(Node *root, std::ostream *out, unsigned maxVars, unsigned long long maxPrintRows) {
76 TruthTableResult res;
77 std::vector<std::string> atoms = collectAtoms(root);
78 res.numVars = atoms.size();
79 if (atoms.size() > maxVars || atoms.size() >= 63) return res; // too large: skipped
80
81 res.computed = true;
82 res.rows = 1ULL << atoms.size();
83 const std::size_t n = atoms.size();
84 std::vector<const Node *> order = postOrderNodes(root);
85 std::vector<char> vals(n), scratch;
86 scratch.reserve(order.size());
87
88 if (out) {
89 for (const std::string &a : atoms) *out << a << '\t';
90 *out << "| result\n";
91 }
92 for (unsigned long long mask = 0; mask < res.rows; ++mask) {
93 for (std::size_t i = 0; i < n; ++i)
94 vals[i] = !((mask >> (n - 1 - i)) & 1ULL); // first row = all T
95 bool r = evaluateOrdered(order, vals, scratch);
96 if (r) ++res.satisfying;
97 if (out && mask < maxPrintRows) {
98 for (std::size_t i = 0; i < n; ++i) *out << (vals[i] ? 'T' : 'F') << '\t';
99 *out << "| " << (r ? 'T' : 'F') << '\n';
100 }
101 }
102 if (out && res.rows > maxPrintRows)
103 *out << "... (" << (res.rows - maxPrintRows) << " more rows not printed)\n";
104 res.satisfiable = res.satisfying > 0;
105 res.tautology = res.satisfying == res.rows;
106 return res;
107}

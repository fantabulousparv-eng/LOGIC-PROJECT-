8CnfInfo readDimacs(const std::string &path) {

12 CnfInfo cnf;
13 std::vector<int> current;
14 std::string line;
15 bool stop = false;
16 while (!stop && std::getline(in, line)) {
17 std::size_t p = line.find_first_not_of(" \t\r");
18 if (p == std::string::npos) continue;
19 char c = line[p];
20 if (c == 'c') continue; // comment
21 if (c == '%') break; // end marker used by some benchmark sets
22 if (c == 'p') { // header: p cnf V C
23 std::istringstream hs(line.substr(p + 1));
24 std::string fmt;
25 std::size_t v = 0;
26 hs >> fmt >> v;
27 cnf.numVars = v;
28 continue;
29 }
30 std::istringstream ls(line);
31 long lit;
32 while (ls >> lit) {
33 if (lit == 0) {
34 if (!current.empty()) cnf.clauses.push_back(current);
35 current.clear();
36 } else {
37 current.push_back(static_cast<int>(lit));
38 }
39 }
40 }
41 if (!current.empty()) cnf.clauses.push_back(current); // last clause without trailing 0
42 cnf.numClauses = cnf.clauses.size();
43 if (cnf.clauses.empty()) throw std::runtime_error("no clauses found in " + path);
44 return cnf;
45}

48static void joinBalanced(const std::vector<std::string> &items, std::size_t lo, std::size_t hi, char op, std::string &out) {
49 if (hi - lo == 1) {
50 out += items[lo];
51 return;
52 }
53 std::size_t mid = lo + (hi - lo) / 2;
54 out += '(';
55 joinBalanced(items, lo, mid, op, out);
56 out += op;
57 joinBalanced(items, mid, hi, op, out);
58 out += ')';

61std::string cnfToInfix(const CnfInfo &cnf) {
62 std::vector<std::string> clauseStrs;
63 clauseStrs.reserve(cnf.clauses.size());
64 for (const auto &clause : cnf.clauses) {
65 std::vector<std::string> lits;
66 lits.reserve(clause.size());
67 for (int l : clause) {
68 std::string name = "x" + std::to_string(std::abs(l));
69 lits.push_back(l < 0 ? "(~" + name + ")" : name);
70 }
71 std::string s;
72 joinBalanced(lits, 0, lits.size(), '+', s);
73 clauseStrs.push_back(std::move(s));
74 }
75 std::string formula;
76 joinBalanced(clauseStrs, 0, clauseStrs.size(), '*', formula);
77 return formula;
78}

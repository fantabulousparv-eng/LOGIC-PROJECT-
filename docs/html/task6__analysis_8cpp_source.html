36static double timeMs(F &&f) {
37 auto t0 = std::chrono::steady_clock::now();
38 f();
39 auto t1 = std::chrono::steady_clock::now();
40 return std::chrono::duration<double, std::milli>(t1 - t0).count();
41}

43FileMetrics analyseFormula(const std::string &name, const std::string &infix, unsigned maxVars, bool useRecursive) {
44 FileMetrics m;
45 m.name = name;
46 m.infixChars = infix.size();
47 Node *root = nullptr;
48 try {
49 auto stage = [&](const std::string &label, double ms) { m.stages.push_back({label, ms}); };
50
51 // Task 1
52 std::string prefix;
53 stage("Task 1: infix to prefix", timeMs([&] { prefix = infixToPrefix(infix); }));
54
55 // Task 2
56 if (useRecursive) {
57 Node *tmp = nullptr;
58 stage("Task 2: prefix to tree (recursive)", timeMs([&] { tmp = prefixToTreeRecursive(prefix); }));
59 freeTree(tmp);
60 }
61 stage("Task 2: prefix to tree (iterative)", timeMs([&] { root = prefixToTreeIterative(prefix); }));
62
63 m.nodes = countNodes(root);
64 m.treeBytes = m.nodes * sizeof(Node);
65
66 // Task 3
67 std::string inR, inI, preR, preI, postR, postI;
68 if (useRecursive) {
69 stage("Task 3: in-order (recursive)", timeMs([&] { inR = inOrderRecursive(root); }));
70 stage("Task 3: pre-order (recursive)", timeMs([&] { preR = preOrderRecursive(root); }));
71 stage("Task 3: post-order (recursive)", timeMs([&] { postR = postOrderRecursive(root); }));
72 }
73 stage("Task 3: in-order (iterative)", timeMs([&] { inI = inOrderIterative(root); }));
74 stage("Task 3: pre-order (iterative)", timeMs([&] { preI = preOrderIterative(root); }));
75 stage("Task 3: post-order (iterative)", timeMs([&] { postI = postOrderIterative(root); }));
76 m.traversalsAgree = useRecursive ? (inR == inI && preR == preI && postR == postI) : true;
77 m.roundTrip = (inI == stripSpaces(infix));
78
79 // Task 4
80 int h1 = 0, h2 = 0;
81 if (useRecursive) stage("Task 4: height (recursive)", timeMs([&] { h1 = heightRecursive(root); }));
82 stage("Task 4: height (iterative)", timeMs([&] { h2 = heightIterative(root); }));
83 m.height = h2;
84 if (useRecursive && h1 != h2) m.traversalsAgree = false;
85
86 // Task 5: one evaluation under a reproducible pseudo-random assignment
87 std::vector<std::string> atoms = collectAtoms(root);
88 m.numVars = atoms.size();
89 std::mt19937 rng(12345);
90 std::vector<char> vals(atoms.size());
91 for (char &v : vals) v = static_cast<char>(rng() & 1U);
92 bool e1 = false, e2 = false;
93 if (useRecursive) stage("Task 5: evaluate (recursive)", timeMs([&] { e1 = evaluateRecursive(root, vals); }));
94 stage("Task 5: evaluate (iterative)", timeMs([&] { e2 = evaluateIterative(root, vals); }));
95 m.evalAgree = useRecursive ? (e1 == e2) : true;
96 m.sampleValue = e2;
97
98 // Task 5: full truth table (only for small formulas)
99 if (atoms.size() <= maxVars)
100 stage("Task 5: full truth table", timeMs([&] { m.table = truthTable(root, nullptr, maxVars); }));
101 else
102 m.table.numVars = atoms.size();
103
104 m.peakRssKb = peakRssKb();
105 m.ok = true;
106 } catch (const std::exception &e) {
107 m.ok = false;
108 m.error = e.what();
109 }
110 freeTree(root);
111 return m;
112}

116static std::string esc(const std::string &s) {
117 std::string r;
118 for (char c : s) {
119 switch (c) {
120 case '&': r += "&amp;"; break;
121 case '<': r += "&lt;"; break;
122 case '>': r += "&gt;"; break;
123 case '"': r += "&quot;"; break;
124 default: r += c;
125 }
126 }
127 return r;
128}
